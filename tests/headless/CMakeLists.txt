cmake_minimum_required(VERSION 3.16)
project(typix_native_layout LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(LV_BUILD_CONF_PATH "${ROOT}/upstream/shared/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo)
if(TYPIX_LVGL_LIBRARY)
 add_library(lvgl STATIC IMPORTED)
 set_target_properties(lvgl PROPERTIES IMPORTED_LOCATION "${TYPIX_LVGL_LIBRARY}")
else()
 add_subdirectory("${ROOT}/upstream/.deps/lvgl" lvgl EXCLUDE_FROM_ALL)
endif()
foreach(app calculator calendar mail)
 string(TOUPPER "${app}" upper)
 add_executable(layout-${app} "${ROOT}/tests/native_layout.cpp" "${ROOT}/tests/headless_transport.cpp" "${ROOT}/port/layout.cpp")
 target_compile_definitions(layout-${app} PRIVATE TEST_${upper})
 target_compile_options(layout-${app} PRIVATE -UNDEBUG)
 target_include_directories(layout-${app} PRIVATE "${ROOT}/port" "${ROOT}/upstream/shared" "${ROOT}/upstream/include" "${ROOT}/upstream/.deps/lvgl" "${ROOT}/upstream/.deps/lvgl/src/libs/tiny_ttf")
 target_link_libraries(layout-${app} PRIVATE lvgl PkgConfig::CAIRO Threads::Threads m)
endforeach()
target_sources(layout-calculator PRIVATE "${ROOT}/upstream/calculator/src/model.cpp" "${ROOT}/upstream/calculator/src/engine.cpp")
target_sources(layout-calendar PRIVATE "${ROOT}/upstream/calendar/src/calendar_model.cpp" "${ROOT}/upstream/calendar/src/calendar_store.cpp")
