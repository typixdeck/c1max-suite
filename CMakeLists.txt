cmake_minimum_required(VERSION 3.16)
project(typix_c1max_suite LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "")
set(UP "${CMAKE_CURRENT_SOURCE_DIR}/upstream")
set(LV_BUILD_CONF_PATH "${UP}/shared/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
add_subdirectory(upstream/.deps/lvgl EXCLUDE_FROM_ALL)
add_library(typix_display STATIC port/display.cpp port/layout.cpp)
target_include_directories(typix_display PUBLIC "${UP}/shared")
target_link_libraries(typix_display PUBLIC lvgl PkgConfig::GTK)
add_library(tinyalsa STATIC upstream/shared/tinyalsa/src/pcm.c upstream/shared/tinyalsa/src/pcm_hw.c upstream/shared/tinyalsa/src/mixer.c upstream/shared/tinyalsa/src/mixer_hw.c upstream/shared/tinyalsa/src/limits.c)
target_include_directories(tinyalsa PUBLIC upstream/shared/tinyalsa/include)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/port" "${UP}/shared" "${UP}/include" "${UP}/.deps/lvgl/src/libs/tiny_ttf" "${UP}/streamplayer/src")
add_library(c1net STATIC upstream/shared/net.cpp upstream/streamplayer/src/client.cpp)
target_link_libraries(c1net PUBLIC pthread)
function(c1_app name)
 add_executable(c1max-${name} ${ARGN})
 target_link_libraries(c1max-${name} PRIVATE typix_display c1net m)
endfunction()
c1_app(calculator upstream/calculator/src/main.cpp upstream/calculator/src/engine.cpp upstream/calculator/src/model.cpp)
c1_app(calendar upstream/calendar/src/main.cpp upstream/calendar/src/calendar_model.cpp upstream/calendar/src/calendar_store.cpp)
c1_app(gomoku upstream/gomoku/src/main.cpp upstream/gomoku/src/game.cpp)
c1_app(airtune upstream/airtune/src/main.cpp)
target_link_libraries(c1max-airtune PRIVATE tinyalsa)
c1_app(streamplayer upstream/streamplayer/src/main.cpp upstream/streamplayer/src/segment_stream.cpp upstream/streamplayer/src/subtitle_stream.cpp upstream/streamplayer/src/subtitles.cpp)
target_link_libraries(c1max-streamplayer PRIVATE tinyalsa)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
add_subdirectory(upstream/.deps/mbedtls-2.28.10 EXCLUDE_FROM_ALL)
c1_app(mail upstream/mail/src/main.cpp upstream/mail/src/protocol.cpp)
target_link_libraries(c1max-mail PRIVATE mbedtls mbedx509 mbedcrypto)
c1_app(bilibili upstream/bilibili/src/main.cpp upstream/bilibili/src/api.cpp upstream/bilibili/src/player.cpp)
target_link_libraries(c1max-bilibili PRIVATE tinyalsa mbedcrypto)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(ENET_NO_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory(upstream/.deps/moonlight-embedded/third_party/moonlight-common-c EXCLUDE_FROM_ALL)
add_library(c1moon STATIC upstream/moonpilot/src/gamestream.cpp upstream/moonpilot/src/remote.cpp upstream/tinyxml2/tinyxml2.cpp)
target_include_directories(c1moon PUBLIC upstream/moonpilot/src PRIVATE upstream/tinyxml2)
target_link_libraries(c1moon PUBLIC c1net moonlight-common-c mbedtls mbedx509 mbedcrypto pthread)
c1_app(moonpilot upstream/moonpilot/src/main.cpp upstream/moonpilot/src/agent.cpp upstream/moonpilot/src/voice.cpp upstream/moonpilot/src/frame_image.cpp)
target_link_libraries(c1max-moonpilot PRIVATE c1moon)
set(QJS "${UP}/.deps/quickjs-2026-06-04")
add_library(c1quickjs STATIC "${QJS}/quickjs.c" "${QJS}/dtoa.c" "${QJS}/libregexp.c" "${QJS}/libunicode.c" "${QJS}/cutils.c")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/qjsinclude")
configure_file("${QJS}/quickjs.h" "${CMAKE_CURRENT_BINARY_DIR}/qjsinclude/quickjs.h" COPYONLY)
target_include_directories(c1quickjs PRIVATE "${QJS}" PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/qjsinclude")
target_compile_definitions(c1quickjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
target_compile_options(c1quickjs PRIVATE -fwrapv)
target_link_libraries(c1quickjs PUBLIC m atomic)
add_library(c1sketch STATIC upstream/processing/src/runtime.cpp)
target_include_directories(c1sketch PUBLIC upstream/processing/src)
target_link_libraries(c1sketch PUBLIC c1quickjs)
c1_app(processing upstream/processing/src/main.cpp)
target_link_libraries(c1max-processing PRIVATE c1sketch)
set(VT "${UP}/terminal/vendor/libvterm")
file(GLOB VTERM_SOURCES "${VT}/src/*.c")
add_library(c1vterm STATIC ${VTERM_SOURCES})
target_include_directories(c1vterm PUBLIC "${VT}/include" PRIVATE "${VT}/src")
c1_app(terminal upstream/terminal/src/main.cpp upstream/terminal/src/terminal.cpp upstream/terminal/src/pty.cpp)
target_link_libraries(c1max-terminal PRIVATE c1vterm)
add_executable(c1max-sketch-test upstream/processing/tests/runtime_test.cpp)
target_link_libraries(c1max-sketch-test PRIVATE c1sketch)
target_compile_options(c1max-sketch-test PRIVATE -UNDEBUG)

# Isolated GUI regressions; never installed in application packages.

target_link_libraries(c1max-escape-streamplayer PRIVATE typix_display c1net tinyalsa m)
